# Host unit tests for the portable firmware modules (no ESP-IDF, nothing downloaded).
#   cmake -S test/host -B build-host && cmake --build build-host && ctest --test-dir build-host --output-on-failure
cmake_minimum_required(VERSION 3.16)
project(caliper_host_tests C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(FW ${CMAKE_CURRENT_LIST_DIR}/../..)

set(PORTABLE_SRCS
    ${FW}/components/measure/measure.c
    ${FW}/components/measure/logrec.c
    ${FW}/components/measure/gatt_codec.c
    ${FW}/components/measure/settings.c
    ${FW}/components/measure/csv.c
    ${FW}/components/measure/buttons.c
    ${FW}/components/measure/hid_keymap.c
    ${FW}/components/sensor_mag/as5311_proto.c
    ${FW}/components/sensor_cap/cap_demod.c
)
set(PORTABLE_INCS
    ${FW}/main/proto
    ${FW}/components/measure/include
    ${FW}/components/sensor_mag/include
    ${FW}/components/sensor_cap/include
)
set(WARN -Wall -Wextra -Werror -Wshadow -Wconversion -Wno-sign-conversion)

add_executable(caliper_tests test_main.c ${PORTABLE_SRCS})
target_include_directories(caliper_tests PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${PORTABLE_INCS})
target_compile_options(caliper_tests PRIVATE ${WARN} -fsanitize=address,undefined -fno-omit-frame-pointer -O1 -g)
target_link_options(caliper_tests PRIVATE -fsanitize=address,undefined)
target_link_libraries(caliper_tests PRIVATE m)

# The portable sources must also build as strict ISO C11 (-std=c11, no GNU/POSIX extensions such as
# strnlen), since the firmware may be built with either. Compiled only; part of the default build.
add_library(portable_c11 OBJECT ${PORTABLE_SRCS})
set_target_properties(portable_c11 PROPERTIES C_EXTENSIONS OFF)
target_include_directories(portable_c11 PRIVATE ${PORTABLE_INCS})
target_compile_options(portable_c11 PRIVATE ${WARN} -pedantic)

# components/store/store.c (the reading log) against small stubs of LittleFS / NVS / FreeRTOS
# (test/host/stubs): the "partition" is a directory in the build tree.
add_executable(store_tests test_store.c ${FW}/components/store/store.c ${PORTABLE_SRCS})
target_include_directories(store_tests PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_LIST_DIR}/stubs
                           ${FW}/components/store/include ${PORTABLE_INCS})
# fsync -> a no-op counter in test_store.c (POSIX fsync per record made the test take ~30 s)
target_compile_definitions(store_tests PRIVATE STORE_HOST_TEST fsync=store_host_fsync "STORE_BASE_PATH=\"${CMAKE_CURRENT_BINARY_DIR}/store_fs\"")
target_compile_options(store_tests PRIVATE ${WARN} -fsanitize=address,undefined -fno-omit-frame-pointer -O1 -g)
target_link_options(store_tests PRIVATE -fsanitize=address,undefined)
target_link_libraries(store_tests PRIVATE m)

# The pin plan (components/board/pin_plan_table.c, generated by tools/pin_plan.py): its invariants.
set(PIN_PLAN_SRCS ${FW}/components/board/pin_plan.c ${FW}/components/board/pin_plan_table.c)
add_executable(pin_plan_tests test_pin_plan.c ${PIN_PLAN_SRCS})
target_include_directories(pin_plan_tests PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${FW}/main ${FW}/components/board/include)
target_compile_definitions(pin_plan_tests PRIVATE PIN_PLAN_HOST)
target_compile_options(pin_plan_tests PRIVATE ${WARN} -fsanitize=address,undefined -fno-omit-frame-pointer -O1 -g)
target_link_options(pin_plan_tests PRIVATE -fsanitize=address,undefined)
add_library(pin_plan_c11 OBJECT ${PIN_PLAN_SRCS})
set_target_properties(pin_plan_c11 PROPERTIES C_EXTENSIONS OFF)
target_include_directories(pin_plan_c11 PRIVATE ${FW}/main ${FW}/components/board/include)
target_compile_definitions(pin_plan_c11 PRIVATE PIN_PLAN_HOST)
target_compile_options(pin_plan_c11 PRIVATE ${WARN} -pedantic)

enable_testing()
add_test(NAME caliper_tests COMMAND caliper_tests)
add_test(NAME store_tests COMMAND store_tests)
add_test(NAME pin_plan_tests COMMAND pin_plan_tests)
# the generated files are current (the ESP-IDF build checks the same at configure time)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME pin_plan_generated COMMAND ${Python3_EXECUTABLE} ${FW}/tools/pin_plan.py --check)
    add_test(NAME board_pins_generated COMMAND ${Python3_EXECUTABLE} ${FW}/tools/gen_pins.py --check)
    add_test(NAME pinstate_exported COMMAND ${Python3_EXECUTABLE} ${FW}/tools/export_pinstate.py --check
             -o ${FW}/test/hwtools/pinstate.json)
endif()
